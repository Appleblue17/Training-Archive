inline int lcpp(int x, int y){
	if(x>y) swap(x, y);
	vector<pii> tem;

				++it2;
			}
		}
	}
	while(!tem.empty()&&tem.back().fi>n-y+1) tem.pop_back();
	int ret=0; 
	for(auto t:tem){
		int w=lcp(x+ret, y+ret);
		if(w<t.fi-ret-1){
			ret+=w;
