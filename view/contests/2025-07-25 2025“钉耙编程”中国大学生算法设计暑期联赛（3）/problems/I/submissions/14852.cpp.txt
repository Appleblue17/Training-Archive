ll ksm(ll a,ll b){
    ll ret=1,base=a;
    while(b>0){
        if(b&1)ret = ret*base%mo;
        base = base*base%mo;
        b/=2;
    }
    return ret%mo;
}

        if(c=='-') flag=1;

    while(c>='0'&&c<='9') x=(x<<1)+(x<<3)+(c-'0'), c=getchar();

}
