int cnt[350][350];

ll sum[350], dt[350];
inline ll qryp(int x){
    return atag[pos[x]]+aself[x];
}
inline ll qry(int x){
    if(x==0) return 0;
    ll ret=sum[pos[x]-1];
    for(int i=lp[pos[x]]; i<=x; ++i){
        ret+=qryp(p[i]);
    }
    return ret;
}
