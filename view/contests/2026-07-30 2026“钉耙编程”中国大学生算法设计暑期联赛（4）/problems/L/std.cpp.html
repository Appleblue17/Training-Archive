8struct SuffixArray {
9  int n;
10  vector<int> sa, rk, lc;
11  SuffixArray(const string& s) : n(SZ(s)), sa(n), rk(n), lc(n - 1) {
12    iota(ALL(sa), 0);
13    sort(ALL(sa), [&](int x, int y) { return s[x] < s[y]; });
14    rk[sa[0]] = 0;
15    for (int i = 1; i < n; i++) {
16      rk[sa[i]] = rk[sa[i - 1]] + (s[sa[i]] != s[sa[i - 1]]);
17    }
18    vector<int> tmp, cnt(n);
19    tmp.reserve(n);
20    for (int w = 1; rk[sa[n - 1]] < n - 1; w *= 2) {
21      tmp.clear();
22      for (int i = n - w; i < n; i++) tmp.pb(i);
23      for (int i = 0; i < n; i++) if (sa[i] >= w) tmp.pb(sa[i] - w);
24      fill(ALL(cnt), 0);
25      for (int i = 0; i < n; i++) cnt[rk[i]]++;
26      partial_sum(ALL(cnt), cnt.begin());
27      for (int i = n - 1; i >= 0; i--) sa[--cnt[rk[tmp[i]]]] = tmp[i];
28      swap(rk, tmp);
29      rk[sa[0]] = 0;
30      for (int i = 1; i < n; i++) {
31        rk[sa[i]] = rk[sa[i - 1]]
32          + (tmp[sa[i]] != tmp[sa[i - 1]] || sa[i - 1] + w == n
33             || tmp[sa[i] + w] != tmp[sa[i - 1] + w]);
34      }
35    }
36    for (int i = 0, k = 0; i < n; i++) {
37      int j = rk[i];
38      if (!j) continue;
39      if (k) k--;
40      while (i + k < n && sa[j - 1] + k < n
41             && s[i + k] == s[sa[j - 1] + k]) k++;
42      lc[j - 1] = k;
43    }
44  }
45};

47struct RMQ {
48  vector<vector<int>> st;
49  RMQ(const vector<int>& a) : st(__lg(SZ(a)) + 1) {
50    st[0] = a;
51    for (int d = 1; d < SZ(st); d++) {
52      int n = SZ(a) - (1 << d) + 1;
53      st[d].resize(n);
54      for (int i = 0; i < n; i++) {
55        st[d][i] = min(st[d - 1][i], st[d - 1][i + (1 << (d - 1))]);
56      }
57    }
58  }
59  int qry(int l, int r) const {
60    int d = __lg(r - l);
61    return min(st[d][l], st[d][r - (1 << d)]);
62  }
63};

65struct PartitionTree {
66  int h;
67  vector<int> V, mid;
68  vector<vector<int>> pre;
69  PartitionTree(vector<int> a) : V(a) {
70    sort(ALL(V));
71    V.erase(unique(ALL(V)), V.end());
72    for (int& x : a) x = int(lower_bound(ALL(V), x) - V.begin());
73    h = 32 - __builtin_clz(max(1, SZ(V) - 1));
74    pre.assign(h, vector<int>(SZ(a) + 1));
75    mid.resize(h);
76    vector<int> b(SZ(a));
77    for (int d = 0; d < h; d++) {
78      int w = h - d - 1;
79      for (int i = 0; i < SZ(a); i++) {
80        pre[d][i + 1] = pre[d][i] + !(a[i] >> w & 1);
81      }
82      int x = 0, y = mid[d] = pre[d].back();
83      for (int v : a) b[v >> w & 1 ? y++ : x++] = v;
84      swap(a, b);
85    }
86  }
87  int less(int l, int r, int v) const {
88    int k = int(lower_bound(ALL(V), v) - V.begin());
89    if (!k) return 0;
90    if (k == SZ(V)) return r - l;
91    int ret = 0;
92    for (int d = 0; d < h; d++) {
93      int ql = pre[d][l], qr = pre[d][r], cnt = qr - ql;
94      if (k >> (h - d - 1) & 1) {
95        ret += cnt;
96        l += mid[d] - ql, r += mid[d] - qr;
97      } else {
98        l = ql, r = qr;
99      }
100    }
101    return ret;
102  }
103  int kth(int l, int r, int k) const {
104    assert(0 <= l && l <= r && 0 <= k && k < r - l);
105    int x = 0;
106    for (int d = 0; d < h; d++) {
107      int ql = pre[d][l], qr = pre[d][r], cnt = qr - ql;
108      if (k < cnt) {
109        l = ql, r = qr;
110      } else {
111        k -= cnt, x |= 1 << (h - d - 1);
112        l += mid[d] - ql, r += mid[d] - qr;
113      }
114    }
115    return V[x];
116  }
117};

119struct DSU {
120  vector<int> f;
121  vector<set<int>> s;
122  vector<vector<pair<int, int>>>& e;
123
124  DSU(const vector<int>& sa, vector<vector<pair<int, int>>>& _e)
125      : f(SZ(sa)), s(SZ(sa)), e(_e) {
126    iota(ALL(f), 0);
127    for (int i = 0; i < SZ(sa); i++) s[i].insert(sa[i]);
128  }
129  int find(int u) {
130    while (u != f[u]) u = f[u] = f[f[u]];
131    return u;
132  }
133  void add(int p, int q, int d) {
134    int x = min(p, q), y = max(p, q);
135    e[x].pb({y, d});
136  }
137  void merge(int u, int v, int d) {
138    u = find(u), v = find(v);
139    if (s[u].size() > s[v].size()) swap(u, v);
140    for (int p : s[u]) {
141      auto it = s[v].lower_bound(p);
142      if (it != s[v].end()) add(p, *it, d);
143      if (it != s[v].begin()) add(p, *prev(it), d);
144    }
145    s[v].merge(s[u]);
146    f[u] = v;
147  }
148};

150struct MaxInfo {
151  int v = -1, w = INT_MAX, x = -1, y = -1;
152  MaxInfo& operator+=(const MaxInfo& o) {
153    if (v < o.v || (v == o.v && tie(o.w, o.x, o.y) < tie(w, x, y))) {
154      *this = o;
155    }
156    return *this;
157  }

161  int v = INT_MAX, w = INT_MAX, x = -1, y = -1;
162  MinInfo& operator+=(const MinInfo& o) {
163    if (tie(o.v, o.w, o.x, o.y) < tie(v, w, x, y)) *this = o;

168template <class T>
169struct BIT {
170  int n;
171  vector<T> a;
172  BIT(int _n = 0) : n(_n), a(_n + 1) {}
173  void mdf(int p, const T& v) {
174    for (int i = p + 1; i <= n; i += i & -i) a[i] += v;
175  }
176  T qry(int p) const {
177    T r{};
178    for (int i = p; i > 0; i -= i & -i) r += a[i];
179    return r;
180  }
181};

183void Main() {
184  int n, m;
185  cin >> n >> m;
186  string s;
187  cin >> s;
188  vector<int> ql(m), qr(m);
189  vector<vector<pair<int, int>>> qs(n);
190  for (int i = 0; i < m; i++) {
191    cin >> ql[i] >> qr[i];
192    ql[i]--;
193    qs[ql[i]].pb({qr[i], i});
194  }
195
196  SuffixArray sa(s);
197  vector<vector<pair<int, int>>> e(n);
198  {
199    vector<pair<int, int>> ed;
200    ed.reserve(n - 1);
201    for (int i = 0; i < n - 1; i++) ed.pb({sa.lc[i], i});
202    sort(ed.rbegin(), ed.rend());
203    DSU dsu(sa.sa, e);
204    for (auto [d, i] : ed) {
205      if (!d) break;
206      dsu.merge(i, i + 1, d);
207    }
208  }
209
210  vector<int> ans_h(m), win(m);
211  {
212    BIT<MaxInfo> bit0(n);
213    BIT<MinInfo> bit1(n);
214    for (int l = n - 1; l >= 0; l--) {
215      for (auto [y, d] : e[l]) {
216        int z = y + d;
217        int w = min(sa.rk[l], sa.rk[y]);
218        bit0.mdf(z - 1, {d, w, l, y});
219        bit1.mdf(n - z, {y, w, l, y});
220      }
221      for (auto [r, id] : qs[l]) {
222        MaxInfo a = bit0.qry(r);
223        MinInfo b = bit1.qry(n - r);
224        int h = a.v, w = a.w, p = a.x;
225        int cur = b.v < r ? r - b.v : -1;
226        if (h < cur || (h == cur && b.w < w)) {
227          h = cur, w = b.w, p = b.x;
228        }
229        assert(h > 0 && p >= 0);
230        ans_h[id] = h, win[id] = p;
231      }
232    }
233  }
234  vector<vector<pair<int, int>>>().swap(e);
235  vector<vector<pair<int, int>>>().swap(qs);
236
237  RMQ rmq(sa.lc);
238  PartitionTree pt(sa.rk);
239  vector<array<int, 3>> ans(m);
240  for (int id = 0; id < m; id++) {
241    int h = ans_h[id], p = sa.rk[win[id]];
242    int l = 0, r = p;
243    while (l < r) {
244      int mid = (l + r) / 2;
245      if (rmq.qry(mid, p) >= h) r = mid;
246      else l = mid + 1;
247    }
248    int A = l;
249    l = p, r = n - 1;
250    while (l < r) {
251      int mid = (l + r + 1) / 2;
252      if (rmq.qry(p, mid) >= h) l = mid;
253      else r = mid - 1;
254    }
255    int B = l;
256
257    int R = qr[id] - h;
258    int x, y;
259    if (A <= sa.rk[R] && sa.rk[R] <= B) {
260      x = R;
261      int k = pt.less(ql[id], R, A);
262      int v = pt.kth(ql[id], R, k);
263      assert(v <= B);
264      y = sa.sa[v];
265    } else {
266      int k = pt.less(ql[id], R + 1, A);
267      int u = pt.kth(ql[id], R + 1, k);
268      int v = pt.kth(ql[id], R + 1, k + 1);
269      assert(u <= B && v <= B);
270      x = sa.sa[u], y = sa.sa[v];
271    }
272    ans[id] = {h, x + 1, y + 1};
273  }
274  for (auto [h, x, y] : ans) cout << h << " " << x << " " << y << "\n";
275}
