13struct BIT {
14  int n;
15  vector<LL> t;
16  BIT(int _n = 0) : n(_n), t(n + 1) {}
17  void add(int p, LL v) {
18    for (p++; p <= n; p += p & -p) t[p] += v;
19  }
20  LL qry(int p) const {
21    LL ret = 0;
22    for (; p; p -= p & -p) ret += t[p];
23    return ret;
24  }
25  LL qry(int l, int r) const { return qry(r) - qry(l); }
26};

28struct RangeMax {
29  int sz;
30  vector<pair<int, int>> t;
31  RangeMax(int n) : sz(1) {
32    while (sz < n) sz *= 2;
33    t.assign(2 * sz, {-1, -1});
34  }
35  void apply(int l, int r, pair<int, int> v) {
36    for (l += sz, r += sz; l < r; l >>= 1, r >>= 1) {
37      if (l & 1) t[l] = max(t[l], v), l++;
38      if (r & 1) --r, t[r] = max(t[r], v);
39    }
40  }
41  int get(int p) const {
42    pair<int, int> ret = {-1, -1};
43    for (p += sz; p; p >>= 1) ret = max(ret, t[p]);
44    return ret.second;
45  }
46};

157  }

168    if (ways[v] >= P) ways[v] -= P;
169    uni[v] = mask | good[v];
