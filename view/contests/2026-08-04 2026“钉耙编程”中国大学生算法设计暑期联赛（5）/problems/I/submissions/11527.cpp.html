137		if(!b){
138			x[dep]=1;

169	pair<int128,int128> solve(int n,int *p,int *m){
170		if(!n) return {0,1};
171		int128 t=p[1],M=m[1];
172		for(int128 i=2;i<=n;i++){
173			int128 a=M,b=m[i],c=(p[i]-t%m[i]+m[i])%m[i];
174			
175			int128 g=exgcd(1,a,b);
176			int128 dx=b/g,dy=a/g;
177			int128 x0=ksj(x[1],c/g,b/g);
178			int128 k=floor(t+M*x0,M*dx);
179			t=t+(x0-k*dx)*M;
180			M=lcm(M,m[i]);
181			t%=M;
182		}
183		return {t,M};
184	}
