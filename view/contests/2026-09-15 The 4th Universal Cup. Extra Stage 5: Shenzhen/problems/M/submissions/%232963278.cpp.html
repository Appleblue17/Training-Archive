200int getTangents(const Point& P, const Circle& C, std::vector<Line>& L) {  
201    Vector u = C.c - P;  
202    double dis = Length(u);

211    return 2;

215int GetLineCircleIntersection(Line& L, const Circle& C, vector<Point>& sol) {  
216    double t1, t2;  
217    double a = L.dir.x, b = L.P.x - C.c.x, c = L.dir.y, d = L.P.y - C.c.y;  
218    double e = a*a + c*c, f = 2.0*(a*b + c*d), g = b*b + d*d - C.r*C.r;  
219    double delta = f*f - 4*e*g; //判别式  
220    if (dcmp(delta) < 0) return 0;  //相离  
221    if (dcmp(delta) == 0) { //相切  
222      t1 = t2 = -f / (2 * e);  
223      sol.push_back(L.point(t1));  
224      return 1;  
225    }  
226    t1 = (-f - sqrt(delta)) / (2.0 * e); sol.push_back(L.point(t1));    // 相交  
227    t2 = (-f + sqrt(delta)) / (2.0 * e); sol.push_back(L.point(t2));  
228    return 2;  
229}
