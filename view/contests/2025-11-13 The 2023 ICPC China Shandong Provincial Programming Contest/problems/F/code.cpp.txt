struct segtree{
    int f[N*4],tag[N*4];
    
    void upd(int o,int x){
        f[o]=f[o]*x%mod;
        tag[o]=tag[o]*x%mod;
    }
    void pushdown(int o){
        upd(o<<1,tag[o]);
        upd(o<<1|1,tag[o]);
        tag[o]=1;
    }
    
    void build(int l,int r,int o){
        f[o]=0,tag[o]=1;
        if(l==r) return ;
        int mid=(l+r)>>1;
        build(l,mid,o<<1);
        build(mid+1,r,o<<1|1);
    }
    
    void mdf(int l,int r,int o,int pos,int x){
        if(l==r){
            gmod(f[o]+=x);
            return ;
        }
        pushdown(o);
        
        int mid=(l+r)>>1;
        if(pos<=mid) mdf(l,mid,o<<1,pos,x);
        else mdf(mid+1,r,o<<1|1,pos,x);
        
        gmod(f[o]=f[o<<1]+f[o<<1|1]);
    }
    
    void modify(int l,int r,int o,int L,int R,int x){
        if(L>R) return ;

            upd(o,x);
            return ;

        pushdown(o);

        if(L<=mid) modify(l,mid,o<<1,L,R,x);
