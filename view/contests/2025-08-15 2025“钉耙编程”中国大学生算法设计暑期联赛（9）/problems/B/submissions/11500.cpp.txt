int ksm(int a,int x){
    int tot=1;
    while(x){
        if(x & 1ll) tot=1ll*tot*a%mod;
        a=1ll*a*a%mod;
        x>>=1ll;
    }
    return tot;
}

struct abc{
    int x,y;
};
abc operator +(abc X,abc Y){
    return (abc){(X.x+Y.x)%mod,(X.y+Y.y)%mod2};
}
abc operator *(abc X,abc Y){
    return (abc){1ll*X.x*Y.x%mod,1ll*X.y*Y.y%mod2};
}
abc operator <<(abc X,abc Y){
    return (abc){1ll*X.x*ksm(2,Y.y)%mod,0};
}
