namespace SIMPLEX {
	
int n,m;  // Number of constraints and variables.
vector< vector<double> > tab;
vector<int> B, N;               // Basic and nonbasic variables.
constexpr double eps = 1e-8;  // Precision.

// Pivot on (N[x], B[y]).
void pivot(int x, int y) {
	swap(N[x], B[y]);
	double v = -1 / tab[x][y];
	for (int j = 0; j <= m + 1; ++j) {
	tab[x][j] = j == y ? -v : v * tab[x][j];
	}
	for (int i = 0; i <= n; ++i) {
	if (i == x) continue;
	v = tab[i][y];
	tab[i][y] = 0;
	for (int j = 0; j <= m + 1; ++j) {
		tab[i][j] += v * tab[x][j];
	}
	}
}

// First phase: find an initial BFS.
// Return false if no feasible solution.
bool initialize() {
	int neg_count = 0;
	for (int j = 0; j < m; ++j) {
	if (tab[n][j] < -eps) {
		for (int i = 0; i <= n; ++i) {
		tab[i][m + 1] += tab[i][j];
		}
		B[j] = ~B[j];
		++neg_count;
	}
	}
	while (neg_count) {
	int x = -1;
	double mi = -eps;
	for (int i = 0; i < n; ++i) {
		if (tab[i][m + 1] < mi) {
		x = i;
		mi = tab[i][m + 1];
		}
	}
	if (x == -1) return false;
	int y = -1;
	mi = INFINITY;
	for (int j = 0; j < m; ++j) {
		if ((B[j] < 0 && tab[x][j] < -eps) || (B[j] >= 0 && tab[x][j] > eps)) {
		auto tmp = tab[n][j] / tab[x][j] + (B[j] < 0 ? -eps : eps);
		if (tmp < mi) {
			y = j;
			mi = tmp;
		}
		}
	}
	if (B[y] < 0) {
		--neg_count;
		B[y] = ~B[y];
		pivot(x, y);
		tab[x][m + 1] += 1;
	} else {
		pivot(x, y);
	}
	}
	return true;
}

// Second phase: find an optimal BFS.
// Return false if the problem is unbounded.
bool simplex() {
	while (true) {
		int x = -1;
		double mi = -eps;
		for (int i = 0; i < n; ++i) {
			if (tab[i][m] < mi) {
				x = i;
				mi = tab[i][m];
			}
		}
		// No column with a negative reduced cost => Optimal.
		if (x == -1) break;
		int y = -1;
		mi = INFINITY;
		for (int j = 0; j < m; ++j) {
			if (tab[x][j] <= eps) continue;
			if (tab[n][j] / tab[x][j] < mi) {
				y = j;
				mi = tab[n][j] / tab[x][j];
			}
		}
		// No row with a positive ratio => Unbounded.
		if (y == -1) return false;
		pivot(x, y);
	}
	return true;
}

// -1 = Infeasible
// 1 = Unbounded
int solve(int n_,int m_) {
	n=n_,m=m_;
	B.resize(m);
	N.resize(n);
	iota(B.begin(), B.end(), n);
	iota(N.begin(), N.end(), 0);
	if(!initialize()) return -1;
	if(simplex()) return 0;
	else return 1;
}

double getmax(){
	return tab[n][m];
}

vector<double> getsol(){
	vector<double> x(n);
	for(int j = 0; j < m; j++) {
		if (B[j] < n) x[B[j]] = tab[n][j];
	}
	return x;
}
