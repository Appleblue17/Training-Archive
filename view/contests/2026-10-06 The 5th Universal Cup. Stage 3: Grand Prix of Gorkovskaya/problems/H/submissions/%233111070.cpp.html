12// Pivot on (N[x], B[y]).
13void pivot(int x, int y) {
14	swap(N[x], B[y]);
15	double v = -1 / tab[x][y];
16	for (int j = 0; j <= m + 1; ++j) {
17	tab[x][j] = j == y ? -v : v * tab[x][j];
18	}

20	if (i == x) continue;

26	}

31bool initialize() {

40	}

43	int x = -1;

56		auto tmp = tab[n][j] / tab[x][j] + (B[j] < 0 ? -eps : eps);
57		if (tmp < mi) {

75// Second phase: find an optimal BFS.
76// Return false if the problem is unbounded.
77bool simplex() {
78	while (true) {
79		int x = -1;
80		double mi = -eps;
81		for (int i = 0; i < n; ++i) {
82			if (tab[i][m] < mi) {
83				x = i;
84				mi = tab[i][m];
85			}
86		}
87		// No column with a negative reduced cost => Optimal.
88		if (x == -1) break;
89		int y = -1;
90		mi = INFINITY;
91		for (int j = 0; j < m; ++j) {
92			if (tab[x][j] <= eps) continue;
93			if (tab[n][j] / tab[x][j] < mi) {
94				y = j;
95				mi = tab[n][j] / tab[x][j];
96			}
97		}
98		// No row with a positive ratio => Unbounded.
99		if (y == -1) return false;
100		pivot(x, y);
101	}
102	return true;
103}
