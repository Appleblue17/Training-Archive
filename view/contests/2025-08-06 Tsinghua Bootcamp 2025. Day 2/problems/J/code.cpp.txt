int ksm(int a,int x,int p){
    int tot=1;
    while(x){
        if(x & 1) tot=(__int128)tot*(__int128)a%p;
        a=(__int128)a*(__int128)a%p;
        x>>=1;

}

int mul[N],inv[N];
void init(int lim){
    mul[0]=inv[0]=1;
    for(int i=1;i<=lim;i++) mul[i]=mul[i-1]*i%mod;

int pollard_rho(int n){

    int c=rng()%(n-1)+1;
    int t=f(0,c,n);
    int r=f(t,c,n);
    while(t!=r){
        int d=__gcd(abs(t-r),n);
        if(d>1) return d;
        t=f(t,c,n);
        r=f(f(r,c,n),c,n);

    return n;
}
