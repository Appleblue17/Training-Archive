int ksm(int a,int x){
    int tot=1;
    while(x){
        if(x & 1ll) tot=tot*a%mod;
        x>>=1ll;
        a=a*a%mod;
    }
    return tot;
}

	void ntt(int *f,int n,int opt){
		for(int i=0;i<=n-1;i++)
			if(i<rev[i]) swap(f[i],f[rev[i]]);
		for(int l=1;l<n;l<<=1){
			int tmp=ksm(3,(mod-1)/(l*2));
			if(opt==-1) tmp=ksm(tmp,mod-2);
			for(int i=0;i<=n-1;i+=l<<1){

				for(int j=0;j<l;j++){

					f[i+j]=(x+y)%mod,f[i+j+l]=(x-y+mod)%mod;
